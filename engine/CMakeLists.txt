# Cada modulo e uma biblioteca INTERFACE (so cabecalhos por enquanto).
# Dependencias apontam sempre para baixo: scene -> events/ecs -> debug -> core.
function(tool_module name)
  cmake_parse_arguments(M "" "" "DEPS" ${ARGN})
  add_library(tool_${name} INTERFACE)
  target_include_directories(tool_${name} INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/${name}/include)
  target_link_libraries(tool_${name} INTERFACE tool_warnings ${M_DEPS})
  add_library(tool::${name} ALIAS tool_${name})
endfunction()

tool_module(core)

# Backend grafico real (Diligent). Vulkan em todas as plataformas, OpenGL no desktop, D3D12 so no Windows.
if(TOOL_WITH_DILIGENT)
  add_library(tool_rhi_diligent STATIC rhi_diligent/src/diligent_device.cpp)
  target_include_directories(tool_rhi_diligent PUBLIC rhi_diligent/include)
  target_link_libraries(tool_rhi_diligent PUBLIC tool_rhi)
  set(_dl_libs Diligent-GraphicsEngineVk-static Diligent-GraphicsEngineOpenGL-static)
  target_compile_definitions(tool_rhi_diligent PRIVATE TOOL_HAS_VULKAN=1 TOOL_HAS_OPENGL=1)
  if(WIN32 AND TOOL_WITH_D3D12)
    target_compile_definitions(tool_rhi_diligent PRIVATE TOOL_HAS_D3D12=1)
    list(APPEND _dl_libs Diligent-GraphicsEngineD3D12-static)
  else()
    target_compile_definitions(tool_rhi_diligent PRIVATE TOOL_HAS_D3D12=0)
  endif()
  target_link_libraries(tool_rhi_diligent PRIVATE ${_dl_libs} Diligent-GraphicsTools Diligent-Common Diligent-GraphicsAccessories Diligent-Primitives Diligent-BasicPlatform Diligent-GraphicsEngine)
  # Alvo da plataforma do Diligent: depende do sistema (antes estava fixo em Linux, o que quebrava no Windows).
  foreach(_p Diligent-Win32Platform Diligent-LinuxPlatform Diligent-AndroidPlatform Diligent-MacPlatform)
    if(TARGET ${_p})
      target_link_libraries(tool_rhi_diligent PRIVATE ${_p})
    endif()
  endforeach()
  add_library(tool::rhi_diligent ALIAS tool_rhi_diligent)
endif()
